# Exceptions for scripts/check-public-data.sh. One rule per line.
#   file:<ERE>  skip files whose path matches
#   <ERE>       skip a finding whose matched value (e-mail, phone, MAC) matches (case-insensitive)
# Every rule needs a comment saying why it is safe.

# Reserved example domains and no-reply senders.
@example\.[a-z]+$
@[a-z0-9.-]*\.example$
noreply
@users\.noreply\.github\.com$

# SSH user of git remotes (git@github.com:owner/repo.git) is not an address.
^git@github\.com$

# systemd unit names such as backup@nightly.service are not addresses.
\.(service|timer|socket)$

# Dummy phone number (all zeros) used in examples and tests.
^\+48[ -]?0{3}[ -]?0{3}[ -]?0{3}$

# Fictional senders used by the Gmail test fixtures (see the 2026-11-17 privacy audit).
file:^ops/elitedesk/tests/test_gmail_(classify|llm)\.py$
